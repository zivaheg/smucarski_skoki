\section{Methodology}

\subsection{Different approaches and error function}

\begin{frame}{Different approaches}
  Different approaches:
  \begin{itemize}
    \item PINN
    \item SSM + PINN (Physics Informed Neural Networks)
    \item pure SSM:
    \begin{itemize}
      \item State vector (x, y, z coordinates and their speed, jumper's speed, and angles)
      \item Observation vector (x, y, z, height)
      \item Controls vector (averaged wind features and angles)
    \end{itemize}
  \end{itemize}

  Average file starter:
  \begin{itemize}
    \item interpolation over the data points 
    \item average number of time stamps
    \item first realistic input of the jump
  \end{itemize}
\end{frame}


\begin{frame}{Error function}
  Error function:
  \begin{itemize}
    \item Error function
    \item leave-one-out cross-validation
  \end{itemize}
  different add-ons to the model:
  \begin{itemize}
    \item all 12 wind sensors or only 3 wind zones
    \item squared wind values
  \end{itemize}
\end{frame}
